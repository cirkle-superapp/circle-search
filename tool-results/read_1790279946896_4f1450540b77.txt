     1→/**
     2→ * search-store.ts
     3→ * -----------------------------------------------------------------------------
     4→ * Client-side state for CIRKLE. Uses Zustand.
     5→ *
     6→ * Responsibilities:
     7→ *   - Hold the current query / mode / filters / results / loading state.
     8→ *   - Execute searches by POSTing to /api/search (relative path — the gateway
     9→ *     forwards it). No absolute URLs anywhere.
    10→ *   - Hydrate from URL params (?q= ?mode= ?freshness= etc.) on mount.
    11→ *   - Push state into URL via window.history.replaceState (NO router.push —
    12→ *     that would cause a full re-render / re-fetch in some Next setups).
    13→ *   - Persist mode + filters (except page) to localStorage when
    14→ *     personalization is ON; in private mode, never persist.
    15→ *   - Drive autocomplete (debounced 200ms GET /api/suggest?q=).
    16→ *   - Open / close the Filters Sheet and the Deep Research Sheet.
    17→ *   - Fetch a source profile for SourceProfileDialog.
    18→ *   - Trigger the seed crawler when the index is empty.
    19→ *   - Load /api/stats for the IndexStatusBar.
    20→ *
    21→ * This module is pure client-side — no Prisma, no LLM client.
    22→ * -----------------------------------------------------------------------------
    23→ */

   167→function isBrowser(): boolean {

   171→/**
   172→ * Push current query/mode/filters state into the URL via history.replaceState.
   173→ * We DO NOT use the Next router — that would trigger a full re-render and
   174→ * potentially re-fetch in some configurations.
   175→ *
   176→ * URL params emitted:
   177→ *   ?q=        query
   178→ *   ?mode=     SearchMode
   179→ *   ?freshness= Freshness
   180→ *   ?src=      comma-separated source types (only if non-empty)
   181→ *   ?diversity= domainDiversity (0|1|2|3)
   182→ *   ?ai=       aiMode
   183→ *   ?pers=     personalization
   184→ *   ?safe=     safeSearch
   185→ *   ?lang=     language (if set)
   186→ *   ?country=  country (if set)
   187→ *   ?page=     page (only if > 1)
   188→ */
   189→function writeUrl(state: { query: string; mode: SearchMode; filters: SearchFilters }) {
   190→  if (!isBrowser()) return
   191→  const params = new URLSearchParams()
   192→  if (state.query.trim()) params.set('q', state.query.trim())
   193→  params.set('mode', state.mode)
   194→  params.set('freshness', state.filters.freshness)
   195→  if (state.filters.sourceTypes.length > 0) {
   196→    params.set('src', state.filters.sourceTypes.join(','))
   197→  }
   198→  params.set('diversity', String(state.filters.domainDiversity))
   199→  params.set('ai', state.filters.aiMode)
   200→  params.set('pers', state.filters.personalization)
   201→  params.set('safe', state.filters.safeSearch)
   202→  if (state.filters.language) params.set('lang', state.filters.language)
   203→  if (state.filters.country) params.set('country', state.filters.country)
   204→  if (state.filters.page > 1) params.set('page', String(state.filters.page))
   205→  if (state.filters.freshness === 'CUSTOM') {
   206→    if (state.filters.freshnessCustomStart) {
   207→      params.set('from', state.filters.freshnessCustomStart)
   208→    }
   209→    if (state.filters.freshnessCustomEnd) {
   210→      params.set('to', state.filters.freshnessCustomEnd)
   211→    }
   212→  }
   213→  const search = params.toString()
   214→  const newUrl = search ? `/?${search}` : '/'
   215→  try {
   216→    window.history.replaceState(null, '', newUrl)
   217→  } catch {
   218→    // ignore — some sandboxed iframes block this
   219→  }
   220→}

   222→function parseFiltersFromUrl(): {
   223→  query: string
   224→  mode: SearchMode
   225→  filters: Partial<SearchFilters>
   226→} {
   227→  if (!isBrowser()) return { query: '', mode: DEFAULT_MODE, filters: {} }
   228→  const params = new URLSearchParams(window.location.search)
   229→  const query = params.get('q') ?? ''
   230→  const modeParam = params.get('mode') as SearchMode | null
   231→  const validModes: SearchMode[] = [
   232→    'BALANCED',
   233→    'EXACT',
   234→    'LATEST',
   235→    'RESEARCH',
   236→    'OFFICIAL',
   237→    'ACADEMIC',
   238→    'COMMUNITY',
   239→    'NEWS',
   240→    'IMAGES',
   241→  ]
   242→  const mode: SearchMode =
   243→    modeParam && validModes.includes(modeParam) ? modeParam : DEFAULT_MODE
   244→  const filters: Partial<SearchFilters> = {}
   245→
   246→  const freshness = params.get('freshness') as Freshness | null
   247→  const validFresh: Freshness[] = ['ANY', 'HOUR', 'DAY', 'WEEK', 'MONTH', 'YEAR', 'CUSTOM']
   248→  if (freshness && validFresh.includes(freshness)) filters.freshness = freshness
   249→
   250→  const src = params.get('src')
   251→  if (src) {
   252→    const arr = src
   253→      .split(',')
   254→      .map((s) => s.trim().toUpperCase())
   255→      .filter((s) =>
   256→        ['OFFICIAL', 'GOVERNMENT', 'ACADEMIC', 'NEWS', 'COMMUNITY', 'COMMERCIAL', 'PRIMARY', 'WEB'].includes(s),
   257→      )
   258→    if (arr.length > 0) filters.sourceTypes = arr as SearchFilters['sourceTypes']
   259→  }
   260→
   261→  const div = params.get('diversity')
   262→  if (div === '0' || div === '1' || div === '2' || div === '3') {
   263→    filters.domainDiversity = parseInt(div, 10) as 0 | 1 | 2 | 3
   264→  }
   265→
   266→  const ai = params.get('ai') as AiMode | null
   267→  if (ai && ['AUTO', 'ON', 'OFF'].includes(ai)) filters.aiMode = ai
   268→
   269→  const pers = params.get('pers') as Personalization | null
   270→  if (pers && ['ON', 'OFF'].includes(pers)) filters.personalization = pers
   271→
   272→  const safe = params.get('safe') as SafeSearch | null
   273→  if (safe && ['ON', 'OFF'].includes(safe)) filters.safeSearch = safe
   274→
   275→  const lang = params.get('lang')
   276→  if (lang) filters.language = lang
   277→
   278→  const country = params.get('country')
   279→  if (country) filters.country = country
   280→
   281→  const page = parseInt(params.get('page') ?? '1', 10)
   282→  if (!Number.isNaN(page) && page > 0) filters.page = page
   283→
   284→  const from = params.get('from')
   285→  const to = params.get('to')
   286→  if (from) filters.freshnessCustomStart = from
   287→  if (to) filters.freshnessCustomEnd = to
   288→
   289→  return { query, mode, filters }
   290→}

   360→      patch.language !== undefined ||

   395→  executeSearch: async () => {
   396→    const state = get()
   397→    const q = state.query.trim()
   398→    if (!q) {
   399→      set({ results: null, error: null })
   400→      get()._writeUrl()
   401→      return
   402→    }
   403→    set({ loading: true, error: null })
   404→    try {
   405→      const body = {
   406→        query: q,
   407→        mode: state.mode,
   408→        filters: state.filters,
   409→      }
   410→      const resp = await fetch('/api/search', {
   411→        method: 'POST',
   412→        headers: { 'Content-Type': 'application/json' },
   413→        body: JSON.stringify(body),
   414→      })

   416→        const text = await resp.text().catch(() => '')
   417→        throw new Error(
   418→          `Search failed (HTTP ${resp.status})${text ? `: ${text.slice(0, 200)}` : ''}`,
   419→        )
   420→      }
   421→      const data = (await resp.json()) as SearchResponse
   422→      set({ results: data, loading: false, aiLayerLoading: true })

   424→      // Record this search in recent-searches (only if personalization is ON —

   435→        loading: false,
   436→        error: e?.message ?? 'Search failed for an unknown reason.',

   568→  // --- Lazy AI layer -------------------------------------------------------
   569→  // Called by executeSearch() AFTER results render. Fetches the AI answer +
   570→  // knowledge card + related questions in parallel via /api/search/ai, then
   571→  // merges them into the existing `results` object WITHOUT re-fetching the
   572→  // organic results. This is what makes the SERP feel fast — results appear
   573→  // in ~1s, AI streams in after.
   574→  loadAILayer: async () => {
   575→    const state = get()
   576→    if (!state.results) return
   577→    const { query, mode, filters, results: currentResults } = state
   578→    if (!query.trim()) return
   579→    // Skip when AI is OFF — no point fetching.
   580→    if (filters.aiMode === 'OFF') {
   581→      set({ aiLayerLoading: false })
   582→      return
   583→    }
   584→    set({ aiLayerLoading: true })
   585→    try {
   586→      const resp = await fetch('/api/search/ai', {
   587→        method: 'POST',
   588→        headers: { 'Content-Type': 'application/json' },
   589→        body: JSON.stringify({
   590→          query,
   591→          mode,
   592→          filters,
   593→          results: currentResults.results.slice(0, 8),
   594→        }),
   595→      })
   596→      if (!resp.ok) return
   597→      const layer = (await resp.json()) as AILayer
   598→      // Merge the AI layer INTO the existing results object (preserve the
   599→      // fast-loaded organic results + instant answer + sponsored + etc).
   600→      const fresh = get().results
   601→      if (!fresh) return
   602→      set({
   603→        results: {
   604→          ...fresh,
   605→          aiAnswer: layer.aiAnswer,
   606→          knowledgeCard: layer.knowledgeCard,
   607→          relatedQuestions: layer.relatedQuestions,
   608→        },
   609→        aiLayerLoading: false,
   610→      })
   611→    } catch {
   612→      set({ aiLayerLoading: false })
   613→    }
   614→  },

   616→  // --- Saved searches (localStorage-only) ---------------------------------
   617→  // Privacy-preserving: never sent to the server. The SavedSearch Prisma
   618→  // model exists for future opt-in sync, but for now everything is local.
   619→
   620→  loadSavedSearches: () => {
   621→    if (!isBrowser()) return
   622→    try {
   623→      const raw = localStorage.getItem(LS_SAVED_KEY)
   624→      if (!raw) {
   625→        set({ savedSearches: [] })
   626→        return
   627→      }
   628→      const parsed = JSON.parse(raw) as SavedSearch[]
   629→      if (Array.isArray(parsed)) {
   630→        set({ savedSearches: parsed })
   631→      }
   632→    } catch {
   633→      set({ savedSearches: [] })
   634→    }
   635→  },
   636→
   637→  saveCurrentSearch: () => {
   638→    const { query, mode, filters, results, savedSearches } = get()
   639→    const q = query.trim()
   640→    if (!q) return
   641→    // Don't save duplicates (same query + mode + freshness + sourceTypes).
   642→    const sig = `${q}|${mode}|${filters.freshness}|${filters.sourceTypes.join(',')}`
   643→    const dupe = savedSearches.find((s) => {
   644→      const s2 = `${s.query}|${s.mode}|${s.filters.freshness}|${s.filters.sourceTypes.join(',')}`
   645→      return s2 === sig
   646→    })
   647→    if (dupe) return
   648→    const entry: SavedSearch = {
   649→      id: `${Date.now()}-${Math.random().toString(36).slice(2, 9)}`,
   650→      query: q,
   651→      mode,
   652→      filters: { ...filters, page: 1 },
   653→      savedAt: new Date().toISOString(),
   654→      resultCount: results?.pagination.totalResults,
   655→    }
   656→    const next = [entry, ...savedSearches].slice(0, 50) // cap at 50
   657→    set({ savedSearches: next })
   658→    if (isBrowser()) {
   659→      try {
   660→        localStorage.setItem(LS_SAVED_KEY, JSON.stringify(next))
   661→      } catch {
   662→        // storage full / unavailable — keep in-memory only
   663→      }
   664→    }
   665→  },
   666→
   667→  deleteSavedSearch: (id) => {
   668→    const next = get().savedSearches.filter((s) => s.id !== id)
   669→    set({ savedSearches: next })
   670→    if (isBrowser()) {
   671→      try {
   672→        localStorage.setItem(LS_SAVED_KEY, JSON.stringify(next))
   673→      } catch {
   674→        // ignore
   675→      }
   676→    }
   677→  },
   678→
   679→  applySavedSearch: (id) => {
   680→    const entry = get().savedSearches.find((s) => s.id === id)
   681→    if (!entry) return
   682→    set({
   683→      query: entry.query,
   684→      mode: entry.mode,
   685→      filters: { ...entry.filters, page: 1 },
   686→      showSavedSearches: false,
   687→    })
   688→    void get().executeSearch()
   689→  },
   690→
   691→  // --- Recent searches (localStorage, only when personalization is ON) ---
   692→  loadRecentSearches: () => {
   693→    if (!isBrowser()) return
   694→    try {
   695→      const raw = localStorage.getItem(LS_RECENT_KEY)
   696→      if (!raw) {
   697→        set({ recentSearches: [] })
   698→        return
   699→      }
   700→      const parsed = JSON.parse(raw) as string[]
   701→      if (Array.isArray(parsed)) set({ recentSearches: parsed })
   702→    } catch {
   703→      set({ recentSearches: [] })
   704→    }
   705→  },
   706→
   707→  recordRecentSearch: (q) => {
   708→    const query = q.trim()
   709→    if (!query) return
   710→    // Privacy: only record when personalization is ON. In private mode,
   711→    // recent searches are never persisted.
   712→    if (get().filters.personalization !== 'ON') return
   713→    const next = [query, ...get().recentSearches.filter((s) => s !== query)].slice(0, 10)
   714→    set({ recentSearches: next })
   715→    if (isBrowser()) {
   716→      try {
   717→        localStorage.setItem(LS_RECENT_KEY, JSON.stringify(next))
   718→      } catch {
   719→        // ignore storage errors
   720→      }
   721→    }
   722→  },
   723→
   724→  clearRecentSearches: () => {
   725→    set({ recentSearches: [] })
   726→    if (isBrowser()) {
   727→      try {
   728→        localStorage.removeItem(LS_RECENT_KEY)
   729→      } catch {
   730→        // ignore
   731→      }
   732→    }
   733→  },
   734→
   735→  hydrateFromUrl: async () => {
   736→    if (get()._hydrated) return
   737→    const { query, mode, filters } = parseFiltersFromUrl()
   738→    const merged: SearchFilters = { ...DEFAULT_FILTERS, ...filters }
   739→    // Always load the personalization preference from localStorage when the
   740→    // URL doesn't explicitly set it — this makes the personalization toggle
   741→    // sticky across page loads (so recent searches work on the home page
   742→    // after the user previously enabled personalization). When the URL DOES
   743→    // set pers=, that wins (URL is the source of truth).
   744→    if (!filters.personalization && isBrowser()) {
   745→      try {
   746→        const saved = JSON.parse(localStorage.getItem(LS_KEY) ?? '{}')
   747→        if (saved.filters?.personalization) {
   748→          merged.personalization = saved.filters.personalization
   749→        }
   750→      } catch {
   751→        // ignore
   752→      }
   753→    }
   754→    // If personalization is ON, also load any saved prefs (mode + structural
   755→    // filters) — but URL wins when both are present.
   756→    if (merged.personalization === 'ON' && isBrowser()) {
   757→      try {
   758→        const saved = JSON.parse(localStorage.getItem(LS_KEY) ?? '{}')
   759→        if (saved.mode && !query) {
   760→          // No query in URL — start from saved prefs.
   761→          ;(set as any)({ mode: saved.mode })
   762→        }
   763→        if (saved.filters && !filters.freshness) {
   764→          Object.assign(merged, saved.filters)
   765→        }
   766→      } catch {
   767→        // ignore
   768→      }
   769→    }
   770→    set({
   771→      query,
   772→      mode: query ? mode : mode, // always honor URL mode if present
   773→      filters: merged,
   774→      _hydrated: true,
   775→    })
   776→    // Persist the resolved personalization preference to localStorage so the
   777→    // NEXT page load (which may not have pers= in the URL) can restore it.
   778→    // This makes the personalization toggle sticky across navigations.
   779→    get()._persistPrefs()
   780→    // Always refresh stats + load any saved/recent searches on hydration.
   781→    void get().loadStats()
   782→    get().loadSavedSearches()
   783→    get().loadRecentSearches()
   784→    if (query.trim()) {
   785→      await get().executeSearch()
   786→    }
   787→  },
   788→
   789→  _writeUrl: () => {
   790→    writeUrl({ query: get().query, mode: get().mode, filters: get().filters })
   791→  },
   792→
   793→  _persistPrefs: () => {
   794→    // Only persist when personalization is ON — in private mode, NEVER persist.
   795→    if (!isBrowser()) return
   796→    const { filters, mode } = get()
   797→    if (filters.personalization !== 'ON') {
   798→      try {
   799→        localStorage.removeItem(LS_KEY)
   800→      } catch {
   801→        // ignore
   802→      }
   803→      return
   804→    }
   805→    try {
   806→      const payload = {
   807→        mode,
   808→        filters: {
   809→          freshness: filters.freshness,
   810→          sourceTypes: filters.sourceTypes,
   811→          domainDiversity: filters.domainDiversity,
   812→          aiMode: filters.aiMode,
   813→          personalization: filters.personalization,
   814→          safeSearch: filters.safeSearch,
   815→          language: filters.language,
   816→          country: filters.country,
   817→          // page intentionally NOT persisted.
   818→        },
   819→      }
   820→      localStorage.setItem(LS_KEY, JSON.stringify(payload))
   821→    } catch {
   822→      // ignore
   823→    }
   824→  },
   825→
   826→  _loadPrefs: () => {
   827→    if (!isBrowser()) return
   828→    try {
   829→      const saved = JSON.parse(localStorage.getItem(LS_KEY) ?? '{}')
   830→      if (saved.mode) set({ mode: saved.mode })
   831→      if (saved.filters) {
   832→        const f = saved.filters
   833→        set({
   834→          filters: {
   835→            ...get().filters,
   836→            ...f,
   837→            page: 1, // never restore page
   838→          },
   839→        })
   840→      }
   841→    } catch {
   842→      // ignore
   843→    }
   844→  },
   845→}))
